# ==============================================================================
#             ULTRA-SMOOTH CUSTOM ELASTIC BOUNCE (PICOM V13 SYNTAX)
# ==============================================================================
animations = (
  {
    triggers = ["open", "show"];
    
    # Scale width with an elastic overshoot curve (1.35 forces the bounce)
    scale-x = {
      curve = "cubic-bezier(0.175, 0.885, 0.32, 1.35)";
      duration = 0.35;
      start = 0.8;
      end = 1.0;
    };
    
    # Scale height with the same curve
    scale-y = {
      curve = "cubic-bezier(0.175, 0.885, 0.32, 1.35)";
      duration = 0.35;
      start = 0.8;
      end = 1.0;
    };
    
    # Smoothly fade the window opacity in parallel
    opacity = {
      curve = "linear";
      duration = 0.25;
      start = 0.0;
      end = 1.0;
    };
  },
  {
    triggers = ["close", "hide"];
    
    # Inward spring pull before vanishing
    scale-x = {
      curve = "cubic-bezier(0.6, -0.28, 0.735, 0.045)";
      duration = 0.2;
      start = 1.0;
      end = 0.8;
    };
    scale-y = {
      curve = "cubic-bezier(0.6, -0.28, 0.735, 0.045)";
      duration = 0.2;
      start = 1.0;
      end = 0.8;
    };
    opacity = {
      curve = "linear";
      duration = 0.2;
      start = 1.0;
      end = 0.0;
    };
  }
);
